### Touch Gesture Detection
```typescript
// Pointer Events API for unified input handling
const handlePointerDown = (e: PointerEvent) => {
  if (e.pointerType === 'touch') {
    // Track for long-press (500ms threshold)
    longPressTimer = setTimeout(() => {
      onLongPress?.(e);
    }, 500);
  }
};

const handlePointerUp = (e: PointerEvent) => {
  clearTimeout(longPressTimer);
  const duration = Date.now() - startTime;

  if (duration < 200 && distance < 10) {
    onTap?.(e); // Quick tap
  }
};
```

### PWA Service Worker
```javascript
// Tier 0: Precache app shell
self.addEventListener('install', (event) => {
  event.waitUntil(
    caches.open('app-shell-v1').then((cache) => {
      return cache.addAll([
        '/',
        '/index.html',
        '/assets/react-vendor-*.js',
        '/assets/game-logic-*.js',
        '/assets/index-*.js',
        '/assets/index-*.css'
      ]);
    })
  );
});

// Tier 1: Cache GeoJSON on-demand
self.addEventListener('fetch', (event) => {
  if (event.request.url.includes('.geojson')) {
    event.respondWith(
      caches.match(event.request).then((cached) => {
        return cached || fetch(event.request).then((response) => {
          return caches.open('map-data-v1').then((cache) => {
            cache.put(event.request, response.clone());
            return response;
          });
        });
      })
    );
  }
});
```

### Bottom Sheet Snap Logic
```typescript
const snapToPoint = (currentY: number) => {
  const snapPoints = [120, window.innerHeight * 0.5, window.innerHeight * 0.85];

  // Find closest snap point
  const closest = snapPoints.reduce((prev, curr) => {
    return Math.abs(curr - currentY) < Math.abs(prev - currentY)
      ? curr
      : prev;
  });

  // Animate with spring physics
  spring.set(closest, {
    stiffness: 300,
    damping: 30
  });
};
```
